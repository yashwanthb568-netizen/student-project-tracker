document.addEventListener('DOMContentLoaded', () => { const form = document.querySelector('#project-form'); const nameInput = document.querySelector('#name'); const statusInput = document.querySelector('#status'); const list = document.querySelector('#project-list'); const emptyState = document.querySelector('#empty-state'); const saveButton = document.querySelector('#save-button'); const cancelButton = document.querySelector('#cancel-button'); // Projects are stored in memory and cleared on page refresh. const projects = []; let editingId = null; let counter = 0; const addLabel = saveButton.textContent; cancelButton.type = 'button'; cancelButton.hidden = true; function createId() { if ( window.crypto && typeof window.crypto.randomUUID === 'function' ) { return window.crypto.randomUUID(); } counter += 1; return 'project-' + Date.now() + '-' + counter; } // Return the form to Add mode. function setAddMode() { editingId = null; form.reset(); nameInput.setCustomValidity(''); saveButton.textContent = addLabel; cancelButton.hidden = true; } // Load a selected project into the form. function startEdit(id) { const project = projects.find((item) => item.id === id); if (!project) { return; } editingId = id; nameInput.value = project.name; statusInput.value = project.status; nameInput.setCustomValidity(''); saveButton.textContent = 'Save changes'; cancelButton.hidden = false; nameInput.focus(); } // Delete only the project matching the selected stable ID. function deleteProject(id) { const project = projects.find((item) => item.id === id); if (!project) { return; } const confirmed = window.confirm( 'Are you sure you want to delete "' + project.name + '"?' ); // Cancel leaves the collection and form unchanged. if (!confirmed) { return; } // If this project is being edited, exit edit mode. if (editingId === id) { setAddMode(); } const index = projects.findIndex( (item) => item.id === id ); if (index === -1) { return; } projects.splice(index, 1); render(); } // Draw the current project collection. function render() { list.textContent = ''; const itemTag = list.tagName === 'UL' || list.tagName === 'OL' ? 'li' : 'div'; projects.forEach((project) => { const item = document.createElement(itemTag); item.dataset.id = project.id; const nameElement = document.createElement('strong'); nameElement.textContent = project.name; const statusElement = document.createElement('span'); statusElement.textContent = ' - ' + project.status + ' '; const editButton = document.createElement('button'); editButton.type = 'button'; editButton.textContent = 'Edit'; editButton.addEventListener('click', () => { startEdit(project.id); }); const deleteButton = document.createElement('button'); deleteButton.type = 'button'; deleteButton.textContent = 'Delete'; deleteButton.addEventListener('click', () => { deleteProject(project.id); }); item.appendChild(nameElement); item.appendChild(statusElement); item.appendChild(editButton); item.appendChild(deleteButton); list.appendChild(item); }); if (emptyState) { emptyState.textContent = projects.length === 0 ? 'No projects added yet.' : ''; emptyState.hidden = projects.length > 0; } } // One submit listener handles both Add and Save changes. form.addEventListener('submit', (event) => { event.preventDefault(); const name = nameInput.value.trim(); const status = statusInput.value; if (name === '') { nameInput.setCustomValidity( 'Please enter a project name.' ); nameInput.reportValidity(); nameInput.focus(); return; } nameInput.setCustomValidity(''); if (editingId === null) { projects.push({ id: createId(), name: name, status: status }); } else { const project = projects.find( (item) => item.id === editingId ); if (project) { project.name = name; project.status = status; } } render(); setAddMode(); nameInput.focus(); }); // Cancel editing without modifying the original project. cancelButton.addEventListener('click', () => { setAddMode(); }); nameInput.addEventListener('input', () => { nameInput.setCustomValidity(''); }); render(); });
